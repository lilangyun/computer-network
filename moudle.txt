// server 主函数
main(argc, argv):
    // 开启日志模块
    init_log()

    // 创建缓冲区
    buf[8192], sbuf[8192]

    // 创建套接字
    socket(sock)
    bind(sock)
    listen(sock)

    // 接收模块
    while true do:
        // 连接
        client_sock <- accept(sock)

        // 接收信息
        while recv(client_sock, buf) >= 1 do:
            // 解析模块
            request <- parse(buf)

            // 响应模块
            keep_alive <- respond_handle(request)
            if keep_alive = 0 then
                break
        // 断开连接
        close(client_sock)
    
    close(sock)
    close_log()

    file_path <- "./static_site/"
    if request.uri = NULL then:
        file_path.cat(index.html)
    else:
        file_path.cat(request.uri)

    // 文件处理
    fd <- open(file_path)
    if fd < 0 then:
        send(response_404)   // 发送 404 Not Found
        error_log(404, file_path)
    data <- read(file_path)  // 文件内容读入data

    // 构建头部
    add_header(request, sbuf, file_path)

    // 发送响应
    send(sbuf)
