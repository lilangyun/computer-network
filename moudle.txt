    init_log()

    // 创建缓冲区
    buf[8192], sbuf[8192]

    // 创建套接字
    socket(sock)
    bind(sock)
    listen(sock)

    // 接收模块
    while true do:
        // 连接
        client_sock <- accept(sock)

        // 接收信息
        while recv(client_sock, buf) >= 1 do:
            // 解析模块
            request <- parse(buf)

            // 响应模块
            keep_alive <- respond_handle(request)
            if keep_alive = 0 then
                break
        // 断开连接
        close(client_sock)
    
    close(sock)
    close_log()

        file_path.cat(request.uri)

    // 文件处理
    fd <- open(file_path)
    if fd < 0 then:
        send(response_404)   // 发送 404 Not Found

// HTTP Pipelining 核心解析逻辑

// 初始化变量
parsed = 0; total_len = 0;

while (keep_alive) {
    // 接收数据
    ret = recv(client_sock, recv_buf + total_len, BUF_SIZE - total_len, 0);
    if (ret <= 0) break;
    total_len += ret;

    // 循环解析缓冲区中的多个请求
    while (parsed < total_len) {
        // 解析一个完整请求，parsed指针自动前移
        req = parse_pipeline(recv_buf, total_len, &parsed);
        
        if (req == NULL) {
            // 跳过错误的请求，继续解析下一个
            skip_to_next_request(recv_buf, &parsed, total_len);
            continue;
        }
        
        // 处理请求并发送响应
        keep_alive = handle_request(req, send_buf, client_sock);
        
        // 释放请求资源
        free(req->headers);
        free(req);
    }
    
    // 所有请求处理完毕，重置缓冲区
    parsed = 0;
    total_len = 0;
}
